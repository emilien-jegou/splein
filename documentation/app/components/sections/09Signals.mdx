let label = count.map(|c| format!("Count: {c}"));

    if pressed.get() { down.get() } else { up.get() }
});

view! {
    <Group on:click={|_| count.update(|c| *c += 1)}>
        <Text size={14.0}>{label}</Text>
    </Group>
}`
