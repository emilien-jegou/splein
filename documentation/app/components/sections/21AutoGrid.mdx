export const code = `// AutoGrid is a first-class element backed by
// dsl::auto_grid::columns_for. Rows are live segments,
// so narrowing reflows and widening merges without a rebuild.
// (Rows arrive as a {vec} splice: control nodes compose
// under <Group>, never as direct component children.)
view! {
    <AutoGrid width={width} min_child={150.0} gap={8.0}>
        {labels
            .get()
            .iter()
            .map(|label| cell(label))
            .collect::<Vec<_>>()}
    </AutoGrid>
}`
